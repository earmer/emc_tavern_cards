  <script type="module">
    function populateCharacterData() {
      const all_variables = getAllVariables();

      // --- 环境数据 ---
      $('#env-date').text(_.get(all_variables, 'stat_data.环境.日期', '－'));
      $('#env-time').text(_.get(all_variables, 'stat_data.环境.时间', '－'));
      $('#env-loc').text(_.get(all_variables, 'stat_data.环境.📍', '－'));

      // --- 角色属性 ---
      const name = _.get(all_variables, 'stat_data.夏夏.基础属性.姓名', '夏星眠');
      const stage = _.get(all_variables, 'stat_data.夏夏.基础属性.特殊阶段', '健康');
      $('#char-name').text(name);
      $('#char-stage').text(stage);

      const stats = {
        'hp': _.get(all_variables, 'stat_data.夏夏.基础属性.体力', 0),
        'hunger': _.get(all_variables, 'stat_data.夏夏.基础属性.饥饿感', 0),
        'mood': _.get(all_variables, 'stat_data.夏夏.情绪.心情', 0),
        'desire': _.get(all_variables, 'stat_data.夏夏.情绪.欲望', 0)
      };

      Object.entries(stats).forEach(([id, val]) => {
        $(`#bar-${id}`).css('width', `${val}%`);
        $(`#val-${id}`).text(`${val}%`);
      });

      // --- 动态 ---
      $('#act-doing').text(_.get(all_variables, 'stat_data.夏夏.动态状态.正在', '－'));
      $('#act-count').text(_.get(all_variables, 'stat_data.夏夏.动态状态.次数', '0'));
      $('#act-outfit').text(_.get(all_variables, 'stat_data.夏夏.动态状态.👗', '－'));

      const debuffs = _.get(all_variables, 'stat_data.夏夏.动态状态.Debuff', {});
      const debuffHtml = Object.entries(debuffs).map(([k, v]) =>
        `<span class="状态标签" title="${v}">${k}</span>`
      ).join('') || '<span style="color:var(--暗文字色); font-size:12px;">常规状态</span>';
      $('#act-debuff').html(debuffHtml);

      // --- 列表填充 ---
      const items = _.get(all_variables, 'stat_data.夏夏.物品栏', {});
      $('#list-items').html(Object.entries(items).map(([n, d]) => `
        <div class="物品条目">
          <span>${n} <small style="color:var(--暗文字色)">(${d.描述 || ''})</small></span>
          <span style="color:var(--柔绿)">×${d.数量 || 1}</span>
        </div>
      `).join('') || '空');

      const history = _.get(all_variables, 'stat_data.夏夏.小记系统.历史小记', {});
      $('#list-history').html(Object.entries(history).reverse().map(([t, c]) => `
        <div style="margin-bottom:8px; border-bottom:1px solid #333; padding-bottom:4px;">
          <div style="font-size:10px; color:var(--柔绿); opacity:0.7;">${t}</div>
          <div style="color:#bbb;">${c}</div>
        </div>
      `).join('') || '无记录');

      $('#note-current').text(_.get(all_variables, 'stat_data.夏夏.小记系统.小记', '－'));
    }

    async function init() {
      await waitGlobalInitialized('Mvu');
      populateCharacterData();
      eventOn(Mvu.events.VARIABLE_UPDATE_ENDED, populateCharacterData);
    }
    $(errorCatched(init));
  </script>
